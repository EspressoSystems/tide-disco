39/// A route handler.
40///
41/// The [Handler] trait defines the interface required of route handlers -- they must be able to
42/// take [RequestParams] and produce either a response or an appropriately typed error.
43///
44/// Implementations of this trait are provided for handler functions returning a serializable
45/// response ([FnHandler]), for boxed handlers, and for the [MapErr] type which can be used to
46/// transform the `Error` type of a [Handler]. This trait is usually used as
47/// `Box<dyn Handler<State, Error>>`, in order to type-erase route-specific details such as the
48/// return type of a handler function. The types which are preserved, `State` and `Error`, should be
49/// the same for all handlers in an API module.
50#[async_trait]
51pub(crate) trait Handler<State, Error>: 'static + Send + Sync {
52    async fn handle(
53        &self,
54        req: RequestParams,
55        state: &State,
56    ) -> Result<tide::Response, RouteError<Error>>;
57}
58
59/// A [Handler] which delegates to an async function.
60///
61/// The function type `F` should be callable as
62/// `async fn(RequestParams, &State) -> Result<R, Error>`. The [Handler] implementation will
63/// automatically convert the result `R` to a [tide::Response] by serializing it, or the error
64/// `Error` to a [RouteError] using [RouteError::AppSpecific]. Note that the format used for
65/// serializing the response is flexible. This implementation will use the [Accept] header of the
66/// incoming request to try to provide a format that the client expects. Supported formats are
67/// `application/json` (using [serde_json]) and `application/octet-stream` (using [bincode]).
68///
69/// # Limitations
70///
71/// [Like many function parameters](crate#boxed-futures) in [tide_disco](crate), the handler
72/// function is required to return a [BoxFuture].
73pub(crate) struct FnHandler<F, VER>(F, PhantomData<VER>);
74
75impl<F, VER> From<F> for FnHandler<F, VER> {
76    fn from(f: F) -> Self {
77        Self(f, Default::default())
78    }
79}

81#[async_trait]
82impl<F, T, State, Error, VER> Handler<State, Error> for FnHandler<F, VER>
83where
84    F: 'static + Send + Sync + Fn(RequestParams, &State) -> BoxFuture<'_, Result<T, Error>>,
85    T: Serialize,
86    State: 'static + Send + Sync,
87    VER: 'static + Send + Sync + StaticVersionType,
88{
89    async fn handle(
90        &self,
91        req: RequestParams,
92        state: &State,
93    ) -> Result<tide::Response, RouteError<Error>> {
94        let accept = req.accept()?;
95        response_from_result(&accept, (self.0)(req, state).await, VER::instance())
96    }
97}

99pub(crate) fn response_from_result<T: Serialize, Error, VER: StaticVersionType>(
100    accept: &Accept,
101    res: Result<T, Error>,
102    bind_version: VER,
103) -> Result<tide::Response, RouteError<Error>> {
104    res.map_err(RouteError::AppSpecific)
105        .and_then(|res| respond_with(accept, &res, bind_version))
106}

121enum RouteImplementation<State, Error> {
122    Http {
123        method: http::Method,
124        handler: Option<Box<dyn Handler<State, Error>>>,
125    },
126    Socket {
127        handler: Option<socket::Handler<State, Error>>,
128    },
129    Metrics {
130        handler: Option<Box<dyn Handler<State, Error>>>,
131    },
132}
133
134impl<State, Error> RouteImplementation<State, Error> {
135    fn map_err<Error2>(
136        self,
137        f: impl 'static + Send + Sync + Fn(Error) -> Error2,
138    ) -> RouteImplementation<State, Error2>
139    where
140        State: 'static + Send + Sync,
141        Error: 'static + Send + Sync,
142        Error2: 'static,
143    {
144        match self {
145            Self::Http { method, handler } => RouteImplementation::Http {
146                method,
147                handler: handler.map(|h| {
148                    let h: Box<dyn Handler<State, Error2>> =
149                        Box::new(MapErr::<Box<dyn Handler<State, Error>>, _, Error>::new(
150                            h, f,
151                        ));
152                    h
153                }),
154            },
155            Self::Socket { handler } => RouteImplementation::Socket {
156                handler: handler.map(|h| socket::map_err(h, f)),
157            },
158            Self::Metrics { handler } => RouteImplementation::Metrics {
159                handler: handler.map(|h| {
160                    let h: Box<dyn Handler<State, Error2>> =
161                        Box::new(MapErr::<Box<dyn Handler<State, Error>>, _, Error>::new(
162                            h, f,
163                        ));
164                    h
165                }),
166            },
167        }
168    }
169}

171/// All the information we need to parse, typecheck, and dispatch a request.
172///
173/// A [Route] is a structured representation of a route specification from an `api.toml` API spec.
174/// It can be parsed from a TOML specification, and it also includes an optional handler function
175/// which the Rust server can register. Routes with no handler will use a default handler that
176/// simply returns information about the route.
177#[derive(Derivative)]
178#[derivative(Debug(bound = ""))]
179pub(crate) struct Route<State, Error> {
180    name: String,
181    patterns: Vec<String>,
182    params: Vec<RequestParam>,
183    doc: String,
184    meta: Arc<ApiMetadata>,
185    #[derivative(Debug = "ignore")]
186    handler: RouteImplementation<State, Error>,
187}
188
189#[derive(Clone, Copy, Debug, Snafu, PartialEq, Eq)]
190pub enum RouteParseError {
191    MissingPathArray,
192    PathElementError,
193    InvalidTypeExpression,
194    UnrecognizedType,
195    MethodMustBeString,
196    InvalidMethod,
197    MissingPath,
198    IncorrectPathType,
199    IncorrectParamType,
200    IncorrectDocType,
201    RouteMustBeTable,
202}
203
204impl<State, Error> Route<State, Error> {
205    /// Parse a [Route] from a TOML specification.
206    ///
207    /// The specification must be a table containing at least the following keys:
208    /// * `PATH`: an array of patterns for this route
209    /// * for each route parameter (route segment starting with `:`), a key with the same name as
210    ///   the parameter (including the `:`) whose value is the type of the parameter
211    ///
212    /// In addition, the following optional keys may be specified:
213    /// * `METHOD`: the method to use to dispatch the route (default `GET`)
214    /// * `DOC`: Markdown description of the route
215    pub fn new(
216        name: String,
217        spec: &toml::Value,
218        meta: Arc<ApiMetadata>,
219    ) -> Result<Self, RouteParseError> {
220        let paths: Vec<String> = spec["PATH"]
221            .as_array()
222            .ok_or(RouteParseError::MissingPathArray)?
223            .iter()
224            .map(|v| {
225                v.as_str()
226                    .ok_or(RouteParseError::PathElementError)
227                    .unwrap()
228                    .to_string()
229            })
230            .collect();
231        let mut params = HashMap::<String, RequestParam>::new();
232        for path in paths.iter() {
233            for seg in path.split('/') {
234                if let Some(name) = seg.strip_prefix(':') {
235                    let param_type = RequestParamType::from_str(
236                        spec[seg]
237                            .as_str()
238                            .ok_or(RouteParseError::InvalidTypeExpression)?,
239                    )
240                    .map_err(|_| RouteParseError::UnrecognizedType)?;
241                    params.insert(
242                        seg.to_string(),
243                        RequestParam {
244                            name: name.to_string(),
245                            param_type,
246                        },
247                    );
248                }
249            }
250        }
251        let method = match spec.get("METHOD") {
252            Some(val) => val
253                .as_str()
254                .context(MethodMustBeStringSnafu)?
255                .parse()
256                .map_err(|_| RouteParseError::InvalidMethod)?,
257            None => Method::get(),
258        };
259        let handler = match method {
260            Method::Http(method) => RouteImplementation::Http {
261                method,
262                handler: None,
263            },
264            Method::Socket => RouteImplementation::Socket { handler: None },
265            Method::Metrics => RouteImplementation::Metrics { handler: None },
266        };
267        Ok(Route {
268            name,
269            patterns: match spec.get("PATH").context(MissingPathSnafu)? {
270                toml::Value::String(s) => vec![s.clone()],
271                toml::Value::Array(paths) => paths
272                    .iter()
273                    .map(|path| Ok(path.as_str().context(IncorrectPathTypeSnafu)?.to_string()))
274                    .collect::<Result<_, _>>()?,
275                _ => return Err(RouteParseError::IncorrectPathType),
276            },
277            params: params.into_values().collect(),
278            handler,
279            doc: match spec.get("DOC") {
280                Some(doc) => markdown::to_html(doc.as_str().context(IncorrectDocTypeSnafu)?),
281                None => String::new(),
282            },
283            meta,
284        })
285    }
286
287    /// The name of the route.
288    ///
289    /// This is the name used to identify the route when binding a handler. It is also the first
290    /// segment of all of the URL patterns for this route.
291    pub fn name(&self) -> String {
292        self.name.clone()
293    }
294
295    /// Iterate over route patterns.
296    pub fn patterns(&self) -> impl Iterator<Item = &String> {
297        self.patterns.iter()
298    }
299
300    /// The HTTP method of the route.
301    pub fn method(&self) -> Method {
302        match &self.handler {
303            RouteImplementation::Http { method, .. } => (*method).into(),
304            RouteImplementation::Socket { .. } => Method::socket(),
305            RouteImplementation::Metrics { .. } => Method::metrics(),
306        }
307    }
308
309    /// Whether a non-default handler has been bound to this route.
310    pub fn has_handler(&self) -> bool {
311        match &self.handler {
312            RouteImplementation::Http { handler, .. } => handler.is_some(),
313            RouteImplementation::Socket { handler, .. } => handler.is_some(),
314            RouteImplementation::Metrics { handler, .. } => handler.is_some(),
315        }
316    }
317
318    /// Get all formal parameters.
319    pub fn params(&self) -> &[RequestParam] {
320        &self.params
321    }
322
323    /// Create a new route with a modified error type.
324    pub fn map_err<Error2>(
325        self,
326        f: impl 'static + Send + Sync + Fn(Error) -> Error2,
327    ) -> Route<State, Error2>
328    where
329        State: 'static + Send + Sync,
330        Error: 'static + Send + Sync,
331        Error2: 'static,
332    {
333        Route {
334            handler: self.handler.map_err(f),
335            name: self.name,
336            patterns: self.patterns,
337            params: self.params,
338            doc: self.doc,
339            meta: self.meta,
340        }
341    }
342
343    /// Compose an HTML fragment documenting all the variations on this route.
344    pub fn documentation(&self) -> Html {
345        html! {
346            (PreEscaped(self.meta.heading_entry
347                .replace("{{METHOD}}", &self.method().to_string())
348                .replace("{{NAME}}", &self.name())))
349            (PreEscaped(&self.meta.heading_routes))
350            @for path in self.patterns() {
351                (PreEscaped(self.meta.route_path.replace("{{PATH}}", &format!("/{}/{}", self.meta.name, path))))
352            }
353            (PreEscaped(&self.meta.heading_parameters))
354            (PreEscaped(&self.meta.parameter_table_open))
355            @for param in self.params() {
356                (PreEscaped(self.meta.parameter_row
357                    .replace("{{NAME}}", &param.name)
358                    .replace("{{TYPE}}", &param.param_type.to_string())))
359            }
360            @if self.params().is_empty() {
361                (PreEscaped(&self.meta.parameter_none))
362            }
363            (PreEscaped(&self.meta.parameter_table_close))
364            (PreEscaped(&self.meta.heading_description))
365            (PreEscaped(&self.doc))
366        }
367    }
368}

370impl<State, Error> Route<State, Error> {
371    pub(crate) fn set_handler(
372        &mut self,
373        h: impl Handler<State, Error>,
374    ) -> Result<(), RouteError<Error>> {
375        match &mut self.handler {
376            RouteImplementation::Http { handler, .. } => {
377                *handler = Some(Box::new(h));
378                Ok(())
379            }
380            _ => Err(RouteError::IncorrectMethod {
381                expected: self.method().to_string(),
382            }),
383        }
384    }
385
386    pub(crate) fn set_fn_handler<F, T, VER>(
387        &mut self,
388        handler: F,
389        _: VER,
390    ) -> Result<(), RouteError<Error>>
391    where
392        F: 'static + Send + Sync + Fn(RequestParams, &State) -> BoxFuture<'_, Result<T, Error>>,
393        T: Serialize,
394        State: 'static + Send + Sync,
395        VER: StaticVersionType + 'static,
396    {
397        self.set_handler(FnHandler::<F, VER>::from(handler))
398    }
399
400    pub(crate) fn set_socket_handler(
401        &mut self,
402        h: socket::Handler<State, Error>,
403    ) -> Result<(), RouteError<Error>> {
404        match &mut self.handler {
405            RouteImplementation::Socket { handler, .. } => {
406                *handler = Some(h);
407                Ok(())
408            }
409            _ => Err(RouteError::IncorrectMethod {
410                expected: self.method().to_string(),
411            }),
412        }
413    }
414
415    pub(crate) fn set_metrics_handler<F, T>(&mut self, h: F) -> Result<(), RouteError<Error>>
416    where
417        F: 'static
418            + Send
419            + Sync
420            + Fn(RequestParams, &State::State) -> BoxFuture<Result<Cow<T>, Error>>,
421        T: 'static + Clone + metrics::Metrics,
422        State: 'static + Send + Sync + ReadState,
423        Error: 'static,
424    {
425        match &mut self.handler {
426            RouteImplementation::Metrics { handler, .. } => {
427                *handler = Some(Box::new(metrics::Handler::from(h)));
428                Ok(())
429            }
430            _ => Err(RouteError::IncorrectMethod {
431                expected: self.method().to_string(),
432            }),
433        }
434    }
435
436    /// Print documentation about the route, to aid the developer when the route is not yet
437    /// implemented.
438    pub(crate) fn default_handler(&self) -> Result<tide::Response, RouteError<Error>> {
439        Ok(tide::Response::builder(StatusCode::NOT_IMPLEMENTED)
440            .body(self.documentation().into_string())
441            .build())
442    }
443
444    pub(crate) async fn handle_socket(
445        &self,
446        req: RequestParams,
447        conn: WebSocketConnection,
448        state: &State,
449    ) -> Result<(), SocketError<Error>> {
450        match &self.handler {
451            RouteImplementation::Socket { handler, .. } => match handler {
452                Some(handler) => handler(req, conn, state).await,
453                // If there is no socket handler registered, the top-level app should register the
454                // route in such a way that the socket upgrade request is rejected and the route
455                // instead displays a simple HTML page with the documentation for the route. That
456                // means that we should not get here.
457                None => unreachable!(),
458            },
459            _ => Err(SocketError::IncorrectMethod {
460                expected: self.method().to_string(),
461                actual: req.method().to_string(),
462            }),
463        }
464    }
465}

467#[async_trait]
468impl<State, Error> Handler<State, Error> for Route<State, Error>
469where
470    Error: 'static,
471    State: 'static + Send + Sync,
472{
473    async fn handle(
474        &self,
475        req: RequestParams,
476        state: &State,
477    ) -> Result<tide::Response, RouteError<Error>> {
478        match &self.handler {
479            RouteImplementation::Http { handler, .. }
480            | RouteImplementation::Metrics { handler, .. } => match handler {
481                Some(handler) => handler.handle(req, state).await,
482                None => self.default_handler(),
483            },
484            RouteImplementation::Socket { .. } => Err(RouteError::IncorrectMethod {
485                expected: self.method().to_string(),
486            }),
487        }
488    }
489}

528/// A special kind of route handler representing a health check.
529///
530/// In order to type-erase the implementation of `HealthCheck` that the handler actually returns,
531/// we store the healthcheck handler in a wrapper that converts the `HealthCheck` to a
532/// `tide::Response`. The status code of the response will be the status code of the `HealthCheck`
533/// implementation returned by the underlying handler.
534pub(crate) type HealthCheckHandler<State> =
535    Box<dyn 'static + Send + Sync + Fn(RequestParams, &State) -> BoxFuture<'_, tide::Response>>;
536
537pub(crate) fn health_check_response<H: HealthCheck, VER: StaticVersionType>(
538    accept: &Accept,
539    health: H,
540) -> tide::Response {
541    let status = health.status();
542    let (body, content_type) =
543        response_body::<H, Infallible, VER>(accept, health).unwrap_or_else(|err| {
544            let msg = format!(
545                "health status was {}, but there was an error generating the response: {}",
546                status, err
547            );
548            (msg.into(), mime::PLAIN)
549        });
550    tide::Response::builder(status)
551        .content_type(content_type)
552        .body(body)
553        .build()
554}
555
556/// A type-erasing wrapper for healthcheck handlers.
557///
558/// Given a handler, this function can be used to derive a new, type-erased [HealthCheckHandler]
559/// that takes only [RequestParams] and returns a generic [tide::Response].
560pub(crate) fn health_check_handler<State, H, VER>(
561    handler: impl 'static + Send + Sync + Fn(&State) -> BoxFuture<H>,
562) -> HealthCheckHandler<State>
563where
564    State: 'static + Send + Sync,
565    H: 'static + HealthCheck,
566    VER: 'static + Send + Sync + StaticVersionType,
567{
568    Box::new(move |req, state| {
569        let accept = req.accept().unwrap_or_else(|_| {
570            // The healthcheck endpoint is not allowed to fail, so just use the default content
571            // type if we can't parse the Accept header.
572            let mut accept = Accept::new();
573            accept.set_wildcard(true);
574            accept
575        });
576        let future = handler(state);
577        async move {
578            let health = future.await;
579            health_check_response::<_, VER>(&accept, health)
580        }
581        .boxed()
582    })
583}

585pub(crate) fn response_body<T: Serialize, E, VER: StaticVersionType>(
586    accept: &Accept,
587    body: T,
588) -> Result<(Body, Mime), RouteError<E>> {
589    let ty = best_response_type(accept, &[mime::JSON, mime::BYTE_STREAM])?;
590    if ty == mime::BYTE_STREAM {
591        let bytes = Serializer::<VER>::serialize(&body).map_err(RouteError::Binary)?;
592        Ok((bytes.into(), mime::BYTE_STREAM))
593    } else if ty == mime::JSON {
594        let json = serde_json::to_string(&body).map_err(RouteError::Json)?;
595        Ok((json.into(), mime::JSON))
596    } else {
597        unreachable!()
598    }
599}

601pub(crate) fn respond_with<T: Serialize, E, VER: StaticVersionType>(
602    accept: &Accept,
603    body: T,
604    _: VER,
605) -> Result<tide::Response, RouteError<E>> {
606    let (body, content_type) = response_body::<_, _, VER>(accept, body)?;
607    Ok(tide::Response::builder(StatusCode::OK)
608        .body(body)
609        .content_type(content_type)
610        .build())
